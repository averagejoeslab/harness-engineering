def stamp(text, start=None):  # when it came in, and how long it ran if it ran

    now = datetime.datetime.now()

    ran = f", ran {(now - start).total_seconds():.1f}s" if start else ""

    return f"[{now:%Y-%m-%d %H:%M:%S}{ran}] {text}"
